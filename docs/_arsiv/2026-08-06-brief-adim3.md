# 1. Kapsama: Pilot'un her içerik satırı bir hedefte mi?

src=open('docs/_arsiv/ocak-pilot-v52.md',encoding='utf-8').read().split('\n')
import glob
top=set()
for p in glob.glob('docs/**/*.md',recursive=True):
    if 'ocak-pilot-v52' in p: continue
    for l in open(p,encoding='utf-8').read().split('\n'):
        if l.strip(): top.add(l)
kayip=[i+1 for i,l in enumerate(src)
       if l.strip() and l.strip()!='---' and not l.startswith('#') and l not in top]
print('KAYIP SATIR:', len(kayip), kayip[:20])
EOF

# 4. Numara tekilliği
awk -F'\t' 'NR>1{print $1}' docs/01-kararlar.tsv | sort -n | uniq -d
